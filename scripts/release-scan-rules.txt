# Safety rules for the release package (read by build-standalone.sh and build-standalone.ps1).
# This scan is a SECOND safeguard. The first is the explicit allowlist of files that are copied at all.
#
#   path: <regex>   matched (case-insensitive, PCRE) against every package-relative path, written with '/'
#   text: <regex>   matched (case-insensitive, PCRE) against the CONTENT of every text file outside bin/,
#                   plus bin/appsettings.json and bin/wwwroot/** (the framework's own files are not scanned)
#
# One rule per line. Blank lines and lines starting with '#' are ignored. A hit fails the build; the report names the
# file, the line number and the rule, never the matched text (it could be a secret).
#
# The rules here are GENERIC on purpose: they describe what internal data looks like (an unknown web host, a network
# share, a non-demo navigation file), not the names of any organisation's systems.
#
# Optional, machine-local: scripts/release-scan-private.txt (same format, git-ignored, never committed). Put the names
# of your own internal systems there, so they are blocked on your machine without being written into the repository.

# --- files that must never be in a package ---
path: \.(db|db-wal|db-shm|sqlite|sqlite3|mdb)$
path: \.(pfx|p12|pem|key|crt|cer|csr|jks|keystore)$
path: \.(pdb|zip|7z|rar|tar|gz|tgz|pdf|mb)$
path: (^|/)server\.ini$
path: (^|/)\.env
path: (^|/)web\.config$
path: (^|/)appsettings\.[A-Za-z]+\.json$

# --- only demonstration data: navigation files must be demo-*.json, AI samples must be CSV ---
path: (^|/)navigation/(?!demo-)[^/]+$
path: (^|/)ai/knowledge/(?!csv(/|$))[^/]+
path: (^|/)ai/prompts/(?!csv(/|$))[^/]+

# --- web addresses: every URL must point at a known-safe host (local, reserved example domains, public documentation) ---
text: https?://(?!(localhost|127\.0\.0\.1|0\.0\.0\.0|\[::1\]|([a-z0-9-]+\.)*example\.(com|org|net)|www\.w3\.org|react\.dev|serilog\.net|api\.openai\.com)(?![A-Za-z0-9-]|\.[A-Za-z0-9]))[A-Za-z0-9._\[-]+

# --- internal host names, network shares and addresses ---
text: \b[a-z0-9-]+(\.[a-z0-9-]+)*\.(gov|mil|edu)(\.[a-z]{2})?\b
text: \b[a-z0-9-]+(\.[a-z0-9-]+)*\.(internal|intranet|corp|lan|localdomain)\b
text: \\\\[A-Za-z0-9_.$-]+\\[A-Za-z0-9_.$-]+
text: (?<![0-9.])(?!(127\.|0\.0\.0\.0))((25[0-5]|2[0-4][0-9]|1[0-9]{2}|[1-9]?[0-9])\.){3}(25[0-5]|2[0-4][0-9]|1[0-9]{2}|[1-9]?[0-9])(?![0-9]|\.[0-9])

# --- credentials and keys ---
text: BEGIN ([A-Z]+ )?PRIVATE KEY
text: \bsk-[A-Za-z0-9]{20,}
text: \bBearer\s+[A-Za-z0-9._~+/-]{20,}
text: (api[_-]?key|passwd|secret)["']?\s*[:=]\s*["']?[A-Za-z0-9/+_.-]{12,}
text: password["']?\s*[:=]\s*["']?(?!change-me|\.\.\.|<)[A-Za-z0-9/+_.-]{8,}
